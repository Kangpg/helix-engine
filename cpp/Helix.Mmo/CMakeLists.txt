add_library(helix_mmo STATIC src/WorldInstance.cpp)
add_library(helix::mmo ALIAS helix_mmo)
target_compile_features(helix_mmo PUBLIC cxx_std_20)
target_link_libraries(helix_mmo PUBLIC Threads::Threads)
target_include_directories(helix_mmo PUBLIC
	$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
	$<INSTALL_INTERFACE:include>)
set_target_properties(helix_mmo PROPERTIES EXPORT_NAME mmo)
if(MSVC)
	target_compile_options(helix_mmo PUBLIC /utf-8)
endif()
install(TARGETS helix_mmo EXPORT helixTargets ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
