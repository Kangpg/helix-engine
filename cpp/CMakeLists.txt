cmake_minimum_required(VERSION 3.21)
project(helix-cpp VERSION 0.1.0 LANGUAGES C CXX)

get_filename_component(HELIX_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
include(${HELIX_SOURCE_ROOT}/cmake/HelixIdeFolders.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

option(HELIX_BUILD_CAPI "Build the optional native C ABI library" OFF)
option(HELIX_BUILD_PROTOCOL "Build Protocol Buffers messages and MMO network sample" ON)
option(HELIX_BUILD_FLATBUFFERS "Build the optional shared FlatBuffers protocol serializer" ON)
option(HELIX_BUILD_BENCHMARKS "Build optional native serialization benchmarks" OFF)
option(HELIX_BUILD_PARALLEL_MMO "Build the experimental C++ Taskflow MMO module" OFF)
option(HELIX_BUILD_SERVICES "Build optional gRPC and Kafka service adapters" OFF)

# 샘플·테스트는 이 저장소를 직접 구성할 때만 기본 ON.
# 다른 프로젝트가 add_subdirectory로 포함하면 라이브러리만 빌드한다.
if(CMAKE_SOURCE_DIR STREQUAL HELIX_SOURCE_ROOT OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	set(_helix_standalone ON)
else()
	set(_helix_standalone OFF)
endif()
option(HELIX_BUILD_SAMPLES "Build sample and demo executables" ${_helix_standalone})
option(HELIX_BUILD_TESTS "Build tests (also requires BUILD_TESTING)" ${_helix_standalone})
set(HELIX_ENABLE_TESTS OFF)
if(HELIX_BUILD_TESTS)
	include(CTest)
	if(BUILD_TESTING)
		set(HELIX_ENABLE_TESTS ON)
	endif()
endif()
if(HELIX_BUILD_PROTOCOL AND (HELIX_BUILD_FLATBUFFERS OR HELIX_BUILD_BENCHMARKS))
	include(${HELIX_SOURCE_ROOT}/cmake/HelixFlatBuffers.cmake)
endif()
add_subdirectory(Helix.Engine)
if(HELIX_BUILD_PROTOCOL)
	add_subdirectory(Helix.Protocol)
	if(HELIX_BUILD_FLATBUFFERS)
		add_subdirectory(Helix.Protocol.FlatBuffers)
	endif()
endif()
add_subdirectory(Helix.Mmo)
if(HELIX_BUILD_SERVICES)
	add_subdirectory(Helix.Services)
endif()
if(HELIX_BUILD_BENCHMARKS)
	if(NOT HELIX_BUILD_PROTOCOL)
		message(FATAL_ERROR "Serialization benchmarks require HELIX_BUILD_PROTOCOL=ON")
	endif()
	add_subdirectory(benchmarks)
endif()

helix_group_ide_targets("${CMAKE_CURRENT_SOURCE_DIR}")
