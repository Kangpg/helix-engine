include(${HELIX_SOURCE_ROOT}/cmake/HelixFlatBuffers.cmake)
set(_generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(_schema "${HELIX_SOURCE_ROOT}/protos/helix_mmo.fbs")
file(MAKE_DIRECTORY "${_generated}")
add_custom_command(OUTPUT "${_generated}/helix_mmo_generated.h"
	COMMAND ${HELIX_FLATC_COMMAND} --cpp -o "${_generated}" "${_schema}"
	DEPENDS "${_schema}" ${HELIX_FLATC_DEPENDENCY} VERBATIM)
add_library(helix_protocol_flatbuffers STATIC src/FlatBuffersSerializer.cpp "${_generated}/helix_mmo_generated.h")
add_library(helix::protocol_flatbuffers ALIAS helix_protocol_flatbuffers)
set_target_properties(helix_protocol_flatbuffers PROPERTIES EXPORT_NAME protocol_flatbuffers)
target_link_libraries(helix_protocol_flatbuffers PUBLIC helix_protocol)
target_include_directories(helix_protocol_flatbuffers PRIVATE "${_generated}" "${helix_flatbuffers_SOURCE_DIR}/include")
target_include_directories(helix_protocol_flatbuffers PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(MSVC)
	target_compile_options(helix_protocol_flatbuffers PRIVATE /utf-8)
endif()
install(TARGETS helix_protocol_flatbuffers EXPORT helixTargets ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES "${_schema}" DESTINATION share/helix/protos)
install(FILES "${helix_flatbuffers_SOURCE_DIR}/LICENSE" DESTINATION share/helix/licenses RENAME flatbuffers-LICENSE)
if(HELIX_ENABLE_TESTS)
	add_executable(helix_serializer_test ../Helix.Protocol/tests/serializer_test.cpp)
	target_link_libraries(helix_serializer_test PRIVATE helix_protocol_flatbuffers)
	target_compile_definitions(helix_serializer_test PRIVATE HELIX_HAS_FLATBUFFERS=1)
	add_test(NAME packet-serializers COMMAND helix_serializer_test)
endif()
